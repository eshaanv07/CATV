
CC = gcc

CFLAGS = -O2 -g -Wall -Wextra -std=gnu11 \
         -Iinclude -Isrc \
         -fcf-protection=none \
         -U_FORTIFY_SOURCE -D_FORTIFY_SOURCE=0

AR = ar
ARFLAGS = rcs

SRC = src/runtime.c src/stack.c
OBJ = $(SRC:.c=.o)

LIB = libcontextchaos.a

TESTS = tests/smoke tests/stack_smoke

.PHONY: all clean test

all: $(LIB)

$(LIB): $(OBJ)
	$(AR) $(ARFLAGS) $@ $^

src/%.o: src/%.c
	$(CC) $(CFLAGS) -c $< -o $@

tests/smoke: tests/smoke.c $(LIB)
	$(CC) $(CFLAGS) $< -L. -lcontextchaos -o $@

tests/stack_smoke: tests/stack_smoke.c $(LIB)
	$(CC) $(CFLAGS) $< -L. -lcontextchaos -o $@

test: $(TESTS)
	./tests/smoke
	./tests/stack_smoke

clean:
	rm -f src/*.o $(LIB) $(TESTS)
